````python
"""Граница §3.2–3.3; локальные дублёры, пока владелец не создал contracts.
````

````python
"""Безопасное чтение и атомарная запись артефактов внутри workspace."""
from __future__ import annotations

import hashlib
import json
import os
import tempfile
from pathlib import Path

from pydantic import BaseModel

from stages.analysis.compat import ContractError


def inside(workspace: Path, relative: str | Path) -> Path:
    base = workspace.resolve()
    candidate = (base / relative).resolve()
    if not candidate.is_relative_to(base):
        raise ContractError("Путь выходит за рабочую папку")
    return candidate


def read_json(workspace: Path, relative: str) -> dict | list:
    try:
        return json.loads(inside(workspace, relative).read_text(encoding="utf-8"))
    except (OSError, ValueError) as exc:
        raise ContractError(f"Не удалось прочитать JSON-артефакт {relative}") from exc


def write_json(workspace: Path, relative: str, value: BaseModel | dict | list) -> str:
    path = inside(workspace, relative)
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = value.model_dump(mode="json", by_alias=True) if isinstance(value, BaseModel) else value
    content = json.dumps(payload, ensure_ascii=False, indent=2, allow_nan=False) + "\n"
    descriptor, temporary = tempfile.mkstemp(prefix=".analysis-", dir=path.parent)
    try:
        with os.fdopen(descriptor, "w", encoding="utf-8") as stream:
            stream.write(content)
        os.replace(temporary, path)
    finally:
        Path(temporary).unlink(missing_ok=True)
    return str(path.relative_to(workspace.resolve()))


def fingerprint(value: str) -> str:
    return hashlib.sha256(value.encode("utf-8")).hexdigest()


def file_fingerprint(path: Path) -> str:
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()
````

````python
"""Детерминированные карточки JSON/CSV/GeoJSON без передачи сырых файлов LLM."""
from __future__ import annotations

import csv
import io
import json
import math
from collections import Counter
from datetime import datetime
from pathlib import Path
from typing import Any

from contracts.analysis import DataCard

MAX_CARD_BYTES = 7000


def _short(value: Any, depth: int = 0) -> Any:
    if depth > 3:
        return "[сокращено]"
    if isinstance(value, str):
        return value[:160] + ("…" if len(value) > 160 else "")
    if isinstance(value, dict):
        return {str(key): _short(item, depth + 1) for key, item in list(value.items())[:12]}
    if isinstance(value, list):
        return [_short(item, depth + 1) for item in value[:6]]
    return value


def _kind(value: Any) -> str:
    if value is None or value == "":
        return "null"
    if isinstance(value, bool):
        return "bool"
    if isinstance(value, int):
        return "int"
    if isinstance(value, float):
        return "decimal"
    if isinstance(value, dict):
        return "object"
    if isinstance(value, list):
        return "array"
    if isinstance(value, str):
        try:
            int(value)
            return "int"
        except ValueError:
            try:
                float(value)
                return "decimal"
            except ValueError:
                pass
        try:
            datetime.fromisoformat(value.replace("Z", "+00:00"))
            return "datetime" if "T" in value or " " in value else "date"
        except ValueError:
            pass
    return "string"


def _records(value: Any, path: str = "$") -> tuple[list, str]:
    if isinstance(value, list):
        return value, path
    if isinstance(value, dict):
        if value.get("type") == "FeatureCollection":
            return value.get("features", []), "$.features"
        candidates: list[tuple[list, str]] = []
        for key, item in value.items():
            if isinstance(item, (list, dict)):
                found, location = _records(item, f"{path}.{key}")
                if found:
                    candidates.append((found, location))
        if candidates:
            return max(candidates, key=lambda item: (any(isinstance(row, dict) for row in item[0]), len(item[0])))
        return [value], path
    return [value], path


def _points(coordinates: Any) -> list[tuple[float, float]]:
    if not isinstance(coordinates, list):
        return []
    if len(coordinates) >= 2 and all(isinstance(v, (int, float)) for v in coordinates[:2]):
        return [(float(coordinates[0]), float(coordinates[1]))]
    return [point for child in coordinates for point in _points(child)]


def _line_length(coordinates: Any) -> float:
    if not isinstance(coordinates, list) or not coordinates:
        return 0.0
    if isinstance(coordinates[0], list) and coordinates[0] and isinstance(coordinates[0][0], (int, float)):
        total = 0.0
        for a, b in zip(coordinates, coordinates[1:]):
            lat1, lat2 = math.radians(a[1]), math.radians(b[1])
            dlat, dlon = lat2 - lat1, math.radians(b[0] - a[0])
            h = math.sin(dlat / 2) ** 2 + math.cos(lat1) * math.cos(lat2) * math.sin(dlon / 2) ** 2
            total += 6371008.8 * 2 * math.asin(min(1, math.sqrt(h)))
        return total
    return sum(_line_length(child) for child in coordinates)


def profile(path: Path, logical_name: str) -> DataCard:
    text = path.read_text(encoding="utf-8-sig")
    if path.suffix.lower() == ".csv":
        try:
            dialect = csv.Sniffer().sniff(text[:8192], delimiters=",;\t|")
        except csv.Error:
            dialect = csv.excel
        reader = csv.DictReader(io.StringIO(text), dialect=dialect)
        if not reader.fieldnames or len(set(reader.fieldnames)) != len(reader.fieldnames):
            raise ValueError("CSV не имеет уникальных заголовков")
        records, location = list(reader), "$rows"
    else:
        def reject_constant(value: str) -> None:
            raise ValueError(f"Недопустимое число JSON: {value}")
        records, location = _records(json.loads(text, parse_constant=reject_constant))
    fields: dict[str, list[Any]] = {}
    geometries: list[dict] = []
    rows: list[dict] = []
    for raw in records:
        row = raw if isinstance(raw, dict) else {"value": raw}
        if row.get("type") == "Feature":
            if isinstance(row.get("geometry"), dict):
                geometries.append(row["geometry"])
            row = dict(row.get("properties") or {})
        rows.append(row)
    for key in sorted({key for row in rows for key in row}):
        fields[key] = [row.get(key) for row in rows]
    details: dict[str, Any] = {}
    anomalies: list[Any] = []
    for key, values in fields.items():
        counts = Counter(json.dumps(value, sort_keys=True, ensure_ascii=False) for value in values)
        kinds = Counter(_kind(value) for value in values)
        empty = sum(value is None or value == "" for value in values)
        def is_zero(value: Any) -> bool:
            if isinstance(value, bool) or _kind(value) not in ("int", "decimal"):
                return False
            return float(value) == 0
        zeros = sum(is_zero(value) for value in values)
        info: dict[str, Any] = {
            "types": dict(kinds), "examples": [_short(value) for value in values[:3]],
            "unique": len(counts), "all_unique": len(counts) == len(values),
            "empty": empty, "zeros": zeros,
            "frequent": [{"value": _short(json.loads(value)), "count": count} for value, count in counts.most_common(3)],
        }
        dates = [str(value) for value in values if _kind(value) in ("date", "datetime")]
        if dates:
            try:
                ordered = sorted(dates, key=lambda v: datetime.fromisoformat(v.replace("Z", "+00:00")).timestamp())
                info["range"] = {"min": ordered[0], "max": ordered[-1]}
            except (ValueError, TypeError, OSError):
                anomalies.append({"field": key, "problem": "Несогласованные часовые пояса или даты"})
        if empty or zeros or len(set(kinds) - {"null"}) > 1:
            anomalies.append({"field": key, "empty": empty, "zeros": zeros, "mixed_types": len(set(kinds) - {"null"}) > 1})
        details[key] = info
    schema: dict[str, Any] = {"records_path": location, "fields": details}
    if geometries:
        points = [p for geometry in geometries for p in _points(geometry.get("coordinates"))]
        lengths = [_line_length(g.get("coordinates")) for g in geometries if g.get("type") in ("LineString", "MultiLineString")]
        schema["geometry"] = {"types": dict(Counter(g.get("type", "unknown") for g in geometries)), "length_unit": "m", "length_method": "haversine_wgs84", "lengths": {"min": min(lengths), "max": max(lengths), "sum": sum(lengths)} if lengths else {}}
        if points:
            schema["geometry"]["bbox"] = [min(p[0] for p in points), min(p[1] for p in points), max(p[0] for p in points), max(p[1] for p in points)]
            if any(abs(x) > 180 or abs(y) > 90 for x, y in points):
                anomalies.append("Координаты вне диапазона WGS84")
    card = DataCard(file=logical_name, records=len(records), schema=schema, anomalies=anomalies, samples=[_short(row) for row in rows[:3]])
    # Ограничиваем именно сериализованную карточку, сохраняя имена всех полей.
    if len(card.model_dump_json().encode()) > MAX_CARD_BYTES:
        for info in details.values():
            info.pop("frequent", None)
            info["examples"] = info["examples"][:1]
        card = DataCard(file=logical_name, records=len(records), schema=schema, anomalies=anomalies[:12], samples=[_short(row) for row in rows[:3]])
    if len(card.model_dump_json().encode()) > MAX_CARD_BYTES:
        card.samples = [{key: value[:32] if isinstance(value, str) else value for key, value in sample.items()} for sample in card.samples]
    if len(card.model_dump_json().encode()) > MAX_CARD_BYTES:
        # Широкий файл не теряет поля: вместо потери схемы явно отклоняем его.
        raise ValueError("Карточка слишком широка для профиля окна; разделите набор по группам полей")
    return card
````

````python
"""S1: текстовые источники, PDF/DOCX и паспорта неразобранных файлов."""
from __future__ import annotations

import re
from pathlib import Path
from typing import Literal

from pydantic import Field

from contracts.analysis import ContractModel, DataCard, Source
from stages.analysis.compat import ContractError
from stages.analysis.profiling import profile
from stages.analysis.storage import inside


class InputFile(ContractModel):
    file: str
    status: Literal["parsed", "partial", "unparsed"]
    reason: str = ""


class InputManifest(ContractModel):
    schema_version: Literal[1] = 1
    files: list[InputFile] = Field(default_factory=list)
    sources: list[Source] = Field(default_factory=list)
    noise: list[str] = Field(default_factory=list)
    data_cards: list[DataCard] = Field(default_factory=list)


def _paragraphs(text: str) -> list[str]:
    return [part.strip() for part in re.split(r"\n\s*\n", text) if part.strip()]


def ingest(requirements: str, attachments: list[str], workspace: Path) -> InputManifest:
    result = InputManifest()

    def add(texts: list[str], name: str) -> None:
        for text in texts:
            if text.strip():
                result.sources.append(Source(id=f"S-{len(result.sources) + 1:03d}", file=name, text=text))

    if requirements.strip():
        add(_paragraphs(requirements), "input/requirements.md")
        result.files.append(InputFile(file="input/requirements.md", status="parsed"))
    seen: set[Path] = set()
    for relative in attachments:
        path = inside(workspace, relative)
        if not path.is_relative_to(inside(workspace, "input")):
            raise ContractError("Приложение должно находиться внутри workspace/input")
        if path in seen or (path.name == "requirements.md" and requirements.strip()):
            continue
        seen.add(path)
        name = str(path.relative_to(workspace.resolve()))
        try:
            suffix = path.suffix.lower()
            if suffix in (".json", ".geojson", ".csv"):
                result.data_cards.append(profile(path, name))
            elif suffix in (".txt", ".md", ".markdown"):
                add(_paragraphs(path.read_text(encoding="utf-8-sig")), name)
            elif suffix == ".docx":
                from docx import Document
                document = Document(path)
                add([p.text for p in document.paragraphs if p.text.strip()], name)
                for table in document.tables:
                    add([" | ".join(cell.text for cell in row.cells) for row in table.rows], name)
                if document.inline_shapes:
                    result.files.append(InputFile(file=name, status="partial", reason="Изображения DOCX не разобраны"))
                    continue
            elif suffix == ".pdf":
                import pymupdf
                with pymupdf.open(path) as document:
                    missing: list[int] = []
                    text_pages = 0
                    for number, page in enumerate(document, 1):
                        text = page.get_text().strip()
                        if text:
                            add([text], name)
                            text_pages += 1
                        if not text or page.get_images():
                            missing.append(number)
                    if missing:
                        result.files.append(InputFile(file=name, status="partial" if text_pages else "unparsed", reason=f"Страницы без текста или с неразобранными изображениями: {missing}"))
                        continue
            else:
                result.files.append(InputFile(file=name, status="unparsed", reason="Формат или изображение пока не поддерживается"))
                continue
            result.files.append(InputFile(file=name, status="parsed"))
        except Exception as exc:
            result.files.append(InputFile(file=name, status="unparsed", reason=f"Не удалось разобрать файл ({type(exc).__name__})"))
    return result
````

````python
from __future__ import annotations

import json
import tempfile
import unittest
from pathlib import Path

import pymupdf
from docx import Document

from stages.analysis.compat import ContractError
from stages.analysis.inputs import ingest
from stages.analysis.profiling import MAX_CARD_BYTES, profile


class InputTests(unittest.TestCase):
    def setUp(self) -> None:
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.root = Path(self.tmp.name)
        (self.root / "input").mkdir()

    def test_text_markdown_stable_fragments(self) -> None:
        result = ingest("# Заголовок\n\nПервый абзац.\n\nВторой абзац.", [], self.root)
        self.assertEqual([source.id for source in result.sources], ["S-001", "S-002", "S-003"])
        self.assertEqual(result.sources[1].text, "Первый абзац.")

    def test_docx_paragraphs_and_table(self) -> None:
        doc = Document()
        doc.add_paragraph("Система должна хранить данные.")
        doc.add_table(rows=1, cols=2).rows[0].cells[0].text = "Заголовок"
        doc.save(self.root / "input/source.docx")
        result = ingest("", ["input/source.docx"], self.root)
        self.assertEqual(result.files[0].status, "parsed")
        self.assertEqual(len(result.sources), 2)

    def test_pdf_text_and_image_slide(self) -> None:
        with pymupdf.open() as pdf:
            pdf.new_page().insert_text((72, 72), "Requirement one")
            pdf.new_page()
            pdf.save(self.root / "input/slides.pdf")
        result = ingest("", ["input/slides.pdf"], self.root)
        self.assertEqual(result.files[0].status, "partial")
        self.assertIn("2", result.files[0].reason)
        self.assertEqual(result.sources[0].text, "Requirement one")

    def test_image_missing_and_malformed_files(self) -> None:
        (self.root / "input/photo.png").write_bytes(b"image")
        (self.root / "input/bad.json").write_text("{broken")
        result = ingest("", ["input/photo.png", "input/missing.txt", "input/bad.json"], self.root)
        self.assertEqual([f.status for f in result.files], ["unparsed"] * 3)

    def test_outside_input_and_symlink_are_rejected(self) -> None:
        for path in ("../private.txt", "product/private.txt"):
            with self.subTest(path=path), self.assertRaises(ContractError):
                ingest("", [path], self.root)
        (self.root / "input/escape").symlink_to(self.root.parent, target_is_directory=True)
        with self.assertRaises(ContractError):
            ingest("", ["input/escape/private.txt"], self.root)

    def test_profile_json_fields_dates_anomalies_and_three_samples(self) -> None:
        rows = [{"id": i, "at": f"2026-10-{i+1:02d}T12:00:00+03:00", "value": None if i == 1 else i, "long": "x" * 4000} for i in range(3)]
        path = self.root / "input/data.json"
        path.write_text(json.dumps({"items": rows}))
        card = profile(path, "input/data.json")
        fields = card.data_schema["fields"]
        self.assertEqual(card.records, 3)
        self.assertEqual(card.data_schema["records_path"], "$.items")
        self.assertTrue(fields["id"]["all_unique"])
        self.assertEqual(fields["value"]["empty"], 1)
        self.assertIn("range", fields["at"])
        self.assertEqual(len(card.samples), 3)
        self.assertLess(len(card.model_dump_json().encode()), MAX_CARD_BYTES)

    def test_csv_delimiter_and_inconsistent_types(self) -> None:
        path = self.root / "input/data.csv"
        path.write_text("id;value\n1;0\n2;missing\n3;\n", encoding="utf-8")
        card = profile(path, path.name)
        self.assertEqual(card.records, 3)
        self.assertEqual(card.data_schema["fields"]["value"]["empty"], 1)
        self.assertTrue(card.anomalies)

    def test_geojson_line_is_not_invented_polygon(self) -> None:
        path = self.root / "input/data.geojson"
        path.write_text(json.dumps({"type": "FeatureCollection", "features": [{"type": "Feature", "properties": {"id": 1}, "geometry": {"type": "LineString", "coordinates": [[37.0, 55.0], [37.01, 55.0]]}}]}))
        card = profile(path, path.name)
        geometry = card.data_schema["geometry"]
        self.assertEqual(geometry["types"], {"LineString": 1})
        self.assertEqual(geometry["bbox"], [37.0, 55.0, 37.01, 55.0])
        self.assertGreater(geometry["lengths"]["sum"], 600)

    unittest.main()
````
