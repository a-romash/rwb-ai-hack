def _initial(manifest: InputManifest) -> dict:
    return {
        "schema_version": 1,
        **{key: [] for key in ("requirements", "roles", "glossary", "issues", "questions", "assumptions", "user_stories", "rules", "acceptance", "nfr", "out_of_scope")},
        "sources": [source.model_dump() for source in manifest.sources],
        "data_cards": [card.model_dump() for card in manifest.data_cards],
        "profile": {"modules": ["core"], "platforms": ["web"]},
    }


def _normalize(text: str) -> str:
    return re.sub(r"\W+", " ", text.casefold()).strip()


def _append(data: dict, section: str, values: list[dict], prefix: str) -> None:
    # Последующие шаги видят уже канонические ID; модель их не перенумеровывает.
    for value in values:
        if section == "acceptance":
            from stages.spec.semantics import normalize_gherkin
            value["gherkin"] = normalize_gherkin(value["gherkin"])
        key = "text" if "text" in value else "want" if "want" in value else "gherkin" if "gherkin" in value else None
        previous = next((item for item in data[section] if key and _normalize(item[key]) == _normalize(value[key])), None)
        if previous is not None:
            for name in ("req", "source", "rules", "stories"):
                if name in value:
                    previous[name] = list(dict.fromkeys(previous.get(name, []) + value[name]))
            continue
        value["id"] = f"{prefix}-{len(data[section]) + 1:02d}"
        data[section].append(value)


def _resolve(data: dict, inp: StageInput) -> list[Question]:
    pending: list[Question] = []
    for question in data["questions"]:
        if question["id"] in inp.answers:
            question["answer"] = inp.answers[question["id"]]
            question["answered_by"] = "human"
        elif question.get("answer") is None and inp.autonomy == "A2":
            question["answer"], question["answered_by"] = question["default"], "default"
        if question.get("answer") is None:
            pending.append(Question.model_validate({key: question[key] for key in ("id", "text", "options", "default", "why", "blocking")}))
        elif not any(assumption["source"] == question["id"] for assumption in data["assumptions"]):
            text = f"{question['text']} Ответ: {question['answer']}."
            if question["answered_by"] == "default":
                text = "Требует подтверждения: " + text
            assumption = Assumption(id=f"ASM-{len(data['assumptions']) + 1:02d}", text=text, source=question["id"], params={})
            data["assumptions"].append(assumption.model_dump())
    return pending


def _findings(analysis: Analysis) -> list[Finding]:
    from gates.g0_inputs import validate_quotes
    from gates.g1_spec import ac_not_testable, br_no_ac, br_no_boundary, ref_unresolved, req_uncovered
    from stages.spec.context import empty_app
    from contracts.app import App
    empty = App.model_validate(empty_app(analysis))
    findings = validate_quotes(analysis)
    for validator in (ref_unresolved, req_uncovered, br_no_ac, br_no_boundary, ac_not_testable):
        findings.extend(validator(analysis, empty))
    return [f for f in findings if f.path.startswith("analysis.")]


def _revise(data: dict, llm: LLM, findings: list[Finding], task_id: str, rejected: list[Finding] | None = None) -> dict:
    normalized_by_path = {}
    for finding in findings:
        address = finding.path.removeprefix("analysis.")
        if address in normalized_by_path:
            normalized_by_path[address].message += " " + finding.message
        else:
            normalized_by_path[address] = finding.model_copy(update={"path": address})
    normalized = list(normalized_by_path.values())
    # Один запрос на находку: в контекст попадает только затронутый элемент.
    for index, finding in enumerate(normalized):
        location = pointer(data, finding.path)
        current = data
        for token in location[1:].split("/"):
            token = token.replace("~1", "/").replace("~0", "~")
            current = current[int(token)] if isinstance(current, list) else current.get(token)
        related = set(re.findall(r"(?:REQ|US|BR|AC|ASM)-\d+", finding.message + (encoded(current) if not isinstance(current, list) else "")))
        for _ in range(3):
            for section in ("rules", "user_stories", "acceptance"):
                for item in data[section]:
                    if item["id"] in related:
                        for name in ("req", "rules", "stories"):
                            related.update(item.get(name, []))
        references = {name: [item for item in data[name] if item.get("id") in related] for name in ("requirements", "rules", "user_stories", "acceptance")}
        references["roles"] = data["roles"]
        if isinstance(current, list):
            current = [{key: value for key, value in item.items() if key in ("id", "req", "rules", "stories")} for item in current]
        sources = {source for req in references["requirements"] for source in req["source"]}
        payload = {"finding": finding.model_dump(), "target_pointer": location, "target_value": current,
                   "section_schema": TypeAdapter(Analysis.model_fields[location.split('/')[1]].annotation).json_schema(),
                   "references": references, "sources": [source for source in data["sources"] if source["id"] in sources] if finding.code == "REQ_NO_QUOTE" else []}
        response = ask(llm, task_class="patch", agent="analyst-repair", payload=payload, output=PatchResponse, task_id=f"{task_id}.{index}")
        try:
            patched = apply_patch(data, response, [finding])
            Analysis.model_validate(patched)
            data = patched
        except (PatchRejected, ValueError, TypeError):
            if rejected is not None:
                rejected.append(finding)
    return data


def execute(inp: StageInput, workspace: Path, llm: LLM, sandbox: Sandbox) -> StageOutput:

    manifest = ingest(inp.requirements, inp.attachments, workspace)
    signature_parts = [inp.requirements, encoded(manifest.model_dump(mode="json"))]
    signature_parts += [file_fingerprint(inside(workspace, name)) for name in inp.attachments if inside(workspace, name).is_file()]
    signature = fingerprint("\n".join(signature_parts))
    data = _initial(manifest)
    flags: list[ElementFlag] = []
    if not manifest.sources:
        flags.append(ElementFlag(path="analysis", status="questioned", reason="Не найдено текстовых бизнес-требований; результаты требуют подтверждения."))

        saved = read_json(workspace, CHECKPOINT)
        if not isinstance(saved, dict):
            raise ContractError("Контрольная точка анализа не соответствует схеме")

            Analysis.model_validate(data)

        write_json(workspace, MANIFEST, manifest)
        if not resumed:

                        pieces.append({"id": source.id, "file": source.file, "text": fragment})

                    pieces.append({"id": source.id, "file": source.file, "text": fragment})

                source_ids = {source["id"] for source in payload["sources"]}
                manifest.noise.extend(source for source in extracted.noise if source in source_ids)
                write_json(workspace, ARTIFACT, Analysis.model_validate(data))
            covered = {source for req in data["requirements"] for source in req["source"]}
            manifest.noise = sorted(set(manifest.noise) - covered)
            if not data["requirements"]:
                flags.append(ElementFlag(path="analysis", status="questioned", reason="Из источников не извлечено ни одного требования; полнота не подтверждена."))
            write_json(workspace, MANIFEST, manifest)
            for index, payload in enumerate(batches(llm, "analyst-domain", Skeleton, data["requirements"], "requirements")):
                skeleton = ask(llm, task_class="req_extract", agent="analyst-domain", payload=payload, output=Skeleton, task_id=f"analysis.domain.{index}")
                for section, key in (("roles", "id"), ("glossary", "term")):
                    for item in getattr(skeleton, section):
                        value = item.model_dump()
                        if not any(old[key] == value[key] for old in data[section]):
                            data[section].append(value)
                for key in ("modules", "platforms"):
                    data["profile"][key] = list(dict.fromkeys((data["profile"][key] if index else []) + getattr(skeleton.profile, key)))
                write_json(workspace, ARTIFACT, Analysis.model_validate(data))
            cards = data["data_cards"]
            lint_base = {"roles": data["roles"], "data_cards": cards}
            for index, payload in enumerate(batches(llm, "analyst-lint", Lint, data["requirements"], "requirements", lint_base)):
                lint = ask(llm, task_class="req_lint", agent="analyst-lint", payload=payload, output=Lint, task_id=f"analysis.lint.{index}")
                mapping: dict[str, str] = {}
                for question in lint.questions:
                    value = question.model_dump()
                    if len(data["questions"]) < 8:
                        value["id"] = f"Q-{len(data['questions']) + 1:02d}"
                        value["answer"] = value["answered_by"] = None
                        mapping[question.id] = value["id"]

                    else:

                    value["id"] = mapping[assumption.id]
                    value["source"] = mapping.get(value["source"], value["source"])

                    data["issues"].append(value)
        if set(inp.answers) - {question["id"] for question in data["questions"]}:
            raise ContractError("Ответы содержат неизвестные идентификаторы вопросов")
        pending = _resolve(data, inp)
        if pending:
            write_json(workspace, CHECKPOINT, {"schema_version": 1, "signature": signature, "phase": "questions", "analysis": data, "manifest": manifest.model_dump(mode="json"), "flags": [flag.model_dump() for flag in flags]})
            write_json(workspace, MANIFEST, manifest)

            return StageOutput(artifacts={"analysis": ARTIFACT, "analysis_inputs": MANIFEST}, summary="Анализ источников выполнен; ожидаются ответы на уточняющие вопросы.", questions=pending, flags=flags)
        base = {"roles": data["roles"], "assumptions": data["assumptions"]}
        declared_roles = [Role.model_validate(role) for role in data["roles"]]

            if section == "user_stories":
                response = stories_schema(declared_roles)
            for index, payload in enumerate(batches(llm, agent, response, data["requirements"], "requirements", base)):
                result = ask(llm, task_class="req_extract", agent=agent, payload=payload, output=response, task_id=f"analysis.{section}.{index}")
                if section == "constraints":
                    _append(data, "nfr", [item.model_dump() for item in result.nfr], "NFR")
                    for item in result.out_of_scope:
                        if item.model_dump() not in data["out_of_scope"]:
                            data["out_of_scope"].append(item.model_dump())
                else:
                    values = [item.model_dump() for item in getattr(result, section)]
                    if section == "user_stories":
                        for value in values:
                            value["role"] = role_id(value["role"], declared_roles)
                    _append(data, section, values, prefix)
                write_json(workspace, ARTIFACT, Analysis.model_validate(data))
        # AC получает только связанные истории/правила одной группы требований.
        for index, payload in enumerate(batches(llm, "analyst-acceptance", Acceptance, data["rules"] + data["user_stories"], "elements", {"roles": data["roles"]})):
            result = ask(llm, task_class="req_extract", agent="analyst-acceptance", payload=payload, output=Acceptance, task_id=f"analysis.acceptance.{index}")
            _append(data, "acceptance", [item.model_dump() for item in result.acceptance], "AC")
            write_json(workspace, ARTIFACT, Analysis.model_validate(data))
        for attempt in range(2):
            issues = _findings(Analysis.model_validate(data))
            if not issues:
                break
            data = _revise(data, llm, issues, f"analysis.revise.{attempt}")
    except BudgetExceeded as exc:
        flags.append(ElementFlag(path="analysis", status="questioned", reason=f"Лимит модели: {exc}; сохранён частичный анализ."))
    except (ValueError, TypeError) as exc:
        raise ContractError("Промежуточный анализ нарушает контракт") from exc
    analysis = Analysis.model_validate(data)
    for issue in _findings(analysis):
        flags.append(ElementFlag(path=issue.path, status="questioned", reason=issue.message, gate="G0" if issue.code == "REQ_NO_QUOTE" else "G1"))
    write_json(workspace, ARTIFACT, analysis)
    write_json(workspace, MANIFEST, manifest)
    write_json(workspace, CHECKPOINT, {"schema_version": 1, "signature": signature, "phase": "complete"})
    return StageOutput(artifacts={"analysis": ARTIFACT, "analysis_inputs": MANIFEST}, summary=f"Разобрано требований: {len(analysis.requirements)}; правил: {len(analysis.rules)}; критериев приёмки: {len(analysis.acceptance)}.", flags=flags)


def repair(inp: StageInput, workspace: Path, llm: LLM, findings: list[Finding]) -> StageOutput:
    validate_input(inp, "analysis")
    data = read_json(workspace, inp.artifacts.get("analysis", ARTIFACT))
    Analysis.model_validate(data)
    original = encoded(data)
    rejected: list[Finding] = []
    try:
        data = _revise(data, llm, findings, f"analysis.repair.{inp.attempt}", rejected)
        flags = [ElementFlag(path=f"analysis.{finding.path}", status="questioned", reason="Предложенный патч отклонён; исходный артефакт сохранён.", gate="G0") for finding in rejected]
    except BudgetExceeded:
        flags = [ElementFlag(path=f.path, status="questioned", reason="Бюджет ремонта исчерпан.", gate="G0") for f in findings]
    write_json(workspace, ARTIFACT, Analysis.model_validate(data))
    summary = "Применён адресный ремонт анализа." if encoded(data) != original else "Изменений в анализ не внесено; требуется повторная проверка гейтом."
    return StageOutput(artifacts={"analysis": ARTIFACT}, summary=summary, flags=flags)

"""Одноходовые запросы, ограничение окна и разбиение узкого контекста."""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, TypeVar

from pydantic import BaseModel

from stages.analysis.compat import BudgetExceeded, ContractError, LLM, StageFailed

ModelT = TypeVar("ModelT", bound=BaseModel)
ROOT = Path(__file__).resolve().parents[2]


def window(llm: LLM) -> int:
    # Шлюз может объявить меньший профиль; этап не меняет параметры генерации.
    return max(1024, min(int(getattr(llm, "context_window", 32768)), 32768))


def encoded(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, separators=(",", ":"), allow_nan=False)


def prompt(agent: str) -> str:
    return (ROOT / "agents" / agent / "prompt.md").read_text(encoding="utf-8")


def size(agent: str, payload: dict, output: type[BaseModel]) -> int:
    # UTF-8 byte count — консервативная верхняя оценка для byte-fallback BPE.
    return len(prompt(agent).encode()) + len(encoded(payload).encode()) + len(encoded(output.model_json_schema()).encode()) + 256


def batches(llm: LLM, agent: str, output: type[BaseModel], items: list[Any], key: str,
            base: dict | None = None, item_limit: int | None = None) -> list[dict]:
    base = base or {}
    groups: list[dict] = []
    current: list[Any] = []
    for item in items:
        candidate = base | {key: current + [item]}
        if current and (size(agent, candidate, output) > window(llm) or (item_limit and len(current) >= item_limit)):
            groups.append(base | {key: current})
            current = []
        if size(agent, base | {key: [item]}, output) > window(llm):
            raise BudgetExceeded("Один элемент контекста превышает профиль окна; требуется более мелкое разбиение")
        current.append(item)
    if current or not items:
        groups.append(base | {key: current})
    return groups


def ask(llm: LLM, *, task_class: str, agent: str, payload: dict,
        output: type[ModelT], task_id: str) -> ModelT:
    if size(agent, payload, output) > window(llm):
        raise BudgetExceeded("Контекст запроса превышает профиль окна")
    # Закрывающий разделитель в недоверенных данных не может закончить блок.
    body = encoded(payload).replace("<", "\\u003c").replace(">", "\\u003e")
    user = "<<<SOURCE\n" + body + "\nSOURCE>>>"
    try:
        value = llm(task_class=task_class, system=prompt(agent), user=user,
                    output=output, task_id=task_id)
        if isinstance(value, BaseModel):
            return output.model_validate(value.model_dump(mode="json", by_alias=True))
        if isinstance(value, str):
            return output.model_validate_json(value)
        return output.model_validate(value)
    except BudgetExceeded:
        raise
    except (ValueError, TypeError) as exc:
        raise ContractError(f"Ответ модели не соответствует схеме {output.__name__}") from exc
    except (TimeoutError, ConnectionError, OSError) as exc:
        raise StageFailed("Шлюз модели недоступен") from exc

"""Внутренние схемы узких шагов; не меняют общие артефакты."""
from typing import Literal

from pydantic import Field, create_model

from contracts.analysis import (
    AcceptanceCriterion, Assumption, BusinessRule, ContractModel, GlossaryEntry,
    Issue, NonFunctionalRequirement, OutOfScope, Profile, Question, Requirement,
    Role, UserStory,

class Extraction(ContractModel):
    requirements: list[Requirement]
    noise: list[str] = Field(default_factory=list)

    roles: list[Role]
    glossary: list[GlossaryEntry]
    profile: Profile

    issues: list[Issue]
    questions: list[Question]
    assumptions: list[Assumption]

    user_stories: list[UserStory]


def stories_schema(roles: list[Role]) -> type[ContractModel]:
    if not roles:
        return Stories
    # Ограничивается внутренний ответ шага, общий контракт Analysis не меняется.
    choices = tuple(dict.fromkeys(value for role in roles for value in (role.id, role.name)))
    story = create_model("StoryForDeclaredRole", __base__=UserStory, role=(Literal[choices], ...))
    return create_model("Stories", __base__=ContractModel, user_stories=(list[story], ...))


class Rules(ContractModel):
    rules: list[BusinessRule]


class Acceptance(ContractModel):
    acceptance: list[AcceptanceCriterion]


class Constraints(ContractModel):
    nfr: list[NonFunctionalRequirement]
    out_of_scope: list[OutOfScope]

"""Разрешение однозначных подписей в уже объявленные идентификаторы ролей."""
from __future__ import annotations

from contracts.analysis import Role


def role_id(value: str, roles: list[Role]) -> str:
    key = value.strip().casefold()
    matches = {role.id for role in roles if key in (role.id.casefold(), role.name.casefold())}
    return next(iter(matches)) if len(matches) == 1 else value


def normalize_section_roles(section: str, value, roles: list[Role]):
    if section == "roles":
        result = {}
        for key, role in value.items():
            canonical = role_id(key, roles)
            if canonical == key:
                canonical = role_id(role["name"], roles)
            result[canonical] = role
        return result
    if section == "permissions":
        for permission in value:
            permission["role"] = role_id(permission["role"], roles)
    elif section in ("commands", "queries"):
        for operation in value.values():
            operation["roles"] = [role_id(role, roles) for role in operation["roles"]]
    elif section == "state_machines":
        for machine in value.values():
            for transition in machine["transitions"]:
                transition["roles"] = [role_id(role, roles) for role in transition.get("roles", [])]
    elif section == "notifications":
        for notification in value.values():
            notification["to"] = role_id(notification["to"], roles)
    return value

import copy
import re
from typing import Literal

from pydantic import Field, JsonValue

from contracts.analysis import ContractModel
from stages.analysis.compat import ContractError, Finding


class PatchRejected(ContractError):
    """Предложенный ремонт отклонён; исходный артефакт остаётся валидным."""


class PatchOperation(ContractModel):
    op: Literal["add", "remove", "replace", "test", "move", "copy"]
    path: str
    value: JsonValue = None
    from_: str | None = Field(default=None, alias="from")


class PatchResponse(ContractModel):
    operations: list[PatchOperation]


def tokens(pointer: str) -> list[str]:
    if not pointer.startswith("/") or re.search(r"~(?![01])", pointer):
        raise PatchRejected("JSON Patch должен использовать корректный JSON Pointer")
    return [token.replace("~1", "/").replace("~0", "~") for token in pointer[1:].split("/")]


def pointer(document: dict, address: str) -> str:
    if address.startswith("/"):
        tokens(address)
        return address
    parts = re.findall(r"[^.\[\]]+", address)
    current = document
    result: list[str] = []
    for part in parts:
        if isinstance(current, list):
            if part.isdecimal():
                index = int(part)
            else:
                matches = [i for i, item in enumerate(current) if isinstance(item, dict) and (item.get("id") == part or item.get("name") == part)]
                if len(matches) != 1:
                    raise PatchRejected("Адрес находки не разрешается в уникальный элемент")
                index = matches[0]
            result.append(str(index))
            current = current[index] if index < len(current) else None
        else:
            result.append(part)
            current = current.get(part) if isinstance(current, dict) else None
    return "/" + "/".join(part.replace("~", "~0").replace("/", "~1") for part in result)


def _at(document: dict, parts: list[str]):
    current = document
    for part in parts:
        if isinstance(current, list):
            if not part.isdecimal() or (len(part) > 1 and part.startswith("0")):
                raise PatchRejected("Недопустимый индекс JSON Pointer")
            current = current[int(part)]
        else:
            current = current[part]
    return current


def apply_patch(document: dict, patch: PatchResponse, findings: list[Finding]) -> dict:
    allowed = [tokens(pointer(document, finding.path)) for finding in findings]
    if not allowed and patch.operations:
        raise PatchRejected("Ремонт без адресов находок запрещён")
    result = copy.deepcopy(document)

    def authorized(parts: list[str]) -> bool:
        return any(parts[:len(root)] == root for root in allowed)

    def remove(parts: list[str]):
        parent = _at(result, parts[:-1])
        if isinstance(parent, list):
            if not parts[-1].isdecimal() or (len(parts[-1]) > 1 and parts[-1].startswith("0")):
                raise PatchRejected("Недопустимый индекс удаления JSON Patch")
            value = parent.pop(int(parts[-1]))
        else:
            value = parent.pop(parts[-1])
        return value

    def put(parts: list[str], value, insert: bool) -> None:
        parent = _at(result, parts[:-1])
        key = parts[-1]
        if parts[0] != "seed":
            if key == "id":
                raise PatchRejected("Изменение стабильного идентификатора запрещено")
            old = None
            if isinstance(parent, dict):
                old = parent.get(key)
            elif not insert and key.isdecimal() and int(key) < len(parent):
                old = parent[int(key)]
            if isinstance(old, dict) and "id" in old and (not isinstance(value, dict) or value.get("id") != old["id"]):
                raise PatchRejected("Замена элемента не должна менять его стабильный идентификатор")
        if isinstance(parent, list):
            if key != "-" and (not key.isdecimal() or (len(key) > 1 and key.startswith("0"))):
                raise PatchRejected("Недопустимый индекс записи JSON Patch")
            index = len(parent) if key == "-" and insert else int(key)
            if index < 0 or index > len(parent) or (not insert and index == len(parent)):
                raise PatchRejected("Индекс JSON Patch вне списка")
            if insert:
                parent.insert(index, value)
            else:
                parent[index] = value
        else:
            if not insert and key not in parent:
                raise PatchRejected("JSON Patch заменяет отсутствующее свойство")
            parent[key] = value

    try:
        for operation in patch.operations:
            parts = tokens(operation.path)
            if not authorized(parts):
                raise PatchRejected("JSON Patch меняет элемент вне адресов находок")
            if len(parts) < 2 and operation.op != "test":
                raise PatchRejected("Замена корня или целого раздела запрещена")
            if parts[-1] == "id" and parts[0] != "seed" and operation.op != "test":
                raise PatchRejected("Изменение стабильного идентификатора запрещено")
            if parts[0] == "requirements" and len(parts) == 2 and operation.op in ("remove", "move"):
                raise PatchRejected("Нельзя скрывать находку удалением исходного требования")
            if operation.op == "test":
                if _at(result, parts) != operation.value:
                    raise PatchRejected("Проверка test в JSON Patch не прошла")
            elif operation.op == "remove":
                remove(parts)
            elif operation.op in ("add", "replace"):
                if "value" not in operation.model_fields_set:
                    raise PatchRejected("Операция JSON Patch не содержит value")
                put(parts, copy.deepcopy(operation.value), operation.op == "add")
            else:
                if operation.from_ is None:
                    raise PatchRejected("Операция JSON Patch не содержит from")
                source = tokens(operation.from_)
                if operation.op == "move" and source[0] == "requirements" and len(source) == 2:
                    raise PatchRejected("Нельзя перемещать исходное требование из анализа")
                if not authorized(source):
                    raise PatchRejected("Источник move/copy находится вне адресов находок")
                if parts[:len(source)] == source and len(parts) > len(source):
                    raise PatchRejected("Нельзя переместить элемент в собственного потомка")
                value = copy.deepcopy(_at(result, source))
                if operation.op == "move":
                    remove(source)
                put(parts, value, True)
    except (KeyError, IndexError, ValueError, TypeError) as exc:
        raise PatchRejected("JSON Patch неприменим к документу") from exc
    return result
````

"""Локальные фейки по §3; не реализуют чужой harness и не вызывают сеть."""
from __future__ import annotations

import copy
import json
from pathlib import Path
from typing import Callable

from pydantic import BaseModel

from stages.analysis.compat import ContractError, StageInput, StageOutput


class FakeLLM:
    def __init__(self, responses: dict, *, context_window: int = 32768) -> None:
        self.responses = copy.deepcopy(responses)
        self.context_window = context_window
        self.calls: list[dict] = []

    @classmethod
    def from_file(cls, path: Path) -> "FakeLLM":
        return cls(json.loads(path.read_text(encoding="utf-8")))

    def __call__(self, *, task_class: str, system: str, user: str,
                 output: type[BaseModel] | str, task_id: str | None = None):
        self.calls.append({"task_class": task_class, "system": system, "user": user, "output": output.__name__ if isinstance(output, type) else output, "task_id": task_id})
        response = self.responses.get(task_id, self.responses.get(task_class))
        if response is None:
            raise ContractError(f"В FakeLLM нет записанного ответа для {task_id}")
        if isinstance(response, Exception):
            raise response
        if callable(response):
            response = response(self.calls[-1])
        return output.model_validate(response) if isinstance(output, type) else str(response)


class FakeSandbox:
    def run(self, *args, **kwargs):
        raise AssertionError("Этапы analysis/spec не должны запускать песочницу")


def run_stage_locally(execute: Callable, *, stage: str, requirements: str,
                      workspace: Path, llm: FakeLLM, artifacts: dict[str, str] | None = None,
                      autonomy: str = "A2", answers: dict[str, str] | None = None) -> StageOutput:
    workspace.mkdir(parents=True, exist_ok=True)
    (workspace / "input").mkdir(exist_ok=True)
    (workspace / "input/requirements.md").write_text(requirements, encoding="utf-8")
    inp = StageInput(run_id="local", stage=stage, attempt=0, requirements=requirements, artifacts=artifacts or {}, autonomy=autonomy, answers=answers or {})
    return execute(inp, workspace, llm, FakeSandbox())
````

from __future__ import annotations

import copy
import json
import tempfile
import unittest
from pathlib import Path

from contracts.analysis import Analysis
from gates.g0_inputs import check as g0
from stages.analysis import execute, repair
from stages.analysis.compat import BudgetExceeded, ContractError, Finding, StageInput
from stages.analysis.runtime import ask, batches, size
from stages.analysis.models import Extraction
from stages.analysis.testing import FakeLLM, run_stage_locally

ROOT = Path(__file__).resolve().parents[3]
RECORDINGS = Path(__file__).parent / "fixtures/tickets-responses.json"


class AnalysisStageTests(unittest.TestCase):
    def setUp(self) -> None:
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.workspace = Path(self.tmp.name)
        self.requirements = (ROOT / "contracts/fixtures/tickets/requirements.md").read_text()
        self.responses = json.loads(RECORDINGS.read_text())

    def run_stage(self, llm: FakeLLM, **kwargs):
        return run_stage_locally(execute, stage="analysis", requirements=self.requirements, workspace=self.workspace, llm=llm, **kwargs)

    def test_a2_completes_with_defaults_and_no_sandbox(self) -> None:
        llm = FakeLLM(self.responses)
        out = self.run_stage(llm)
        self.assertEqual(out.questions, [])
        self.assertEqual(out.flags, [])
        self.assertTrue(g0(self.workspace, out.artifacts).passed)
        analysis = Analysis.model_validate_json((self.workspace / "spec/analysis.json").read_text())
        self.assertTrue(all(q.answered_by == "default" for q in analysis.questions))
        self.assertEqual(len(llm.calls), 7)
        self.assertTrue(all("<<<SOURCE\n" in call["user"] for call in llm.calls))
        self.assertTrue(all("messages" not in call for call in llm.calls))

    def test_a1_checkpoint_resumes_without_reextracting(self) -> None:
        llm = FakeLLM(self.responses)
        waiting = self.run_stage(llm, autonomy="A1")
        self.assertEqual(len(waiting.questions), 3)
        self.assertEqual(len(llm.calls), 3)
        resumed = FakeLLM({key: value for key, value in self.responses.items() if key not in ("analysis.extract.0", "analysis.domain.0", "analysis.lint.0")})
        answers = {question.id: question.default for question in waiting.questions}
        out = self.run_stage(resumed, autonomy="A1", answers=answers)
        self.assertEqual(out.questions, [])
        self.assertEqual(len(resumed.calls), 4)
        analysis = Analysis.model_validate_json((self.workspace / "spec/analysis.json").read_text())
        self.assertTrue(all(q.answered_by == "human" for q in analysis.questions))

    def test_partial_answers_remain_waiting(self) -> None:
        waiting = self.run_stage(FakeLLM(self.responses), autonomy="A0")
        out = self.run_stage(FakeLLM({}), autonomy="A0", answers={waiting.questions[0].id: "Особый календарь"})
        self.assertEqual(len(out.questions), 2)

    def test_budget_exhaustion_preserves_schema_valid_partial_artifact(self) -> None:
        self.responses["analysis.domain.0"] = BudgetExceeded()
        out = self.run_stage(FakeLLM(self.responses))
        self.assertTrue(out.flags)
        analysis = Analysis.model_validate_json((self.workspace / "spec/analysis.json").read_text())
        self.assertEqual(len(analysis.requirements), 9)
        self.assertEqual(analysis.user_stories, [])

    def test_malformed_response_is_contract_error(self) -> None:
        self.responses["analysis.extract.0"] = {"unexpected": True}
        with self.assertRaises(ContractError):
            self.run_stage(FakeLLM(self.responses))

    def test_invalid_quote_is_repaired_as_targeted_patch(self) -> None:
        quote = self.responses["analysis.extract.0"]["requirements"][0]["quote"]
        self.responses["analysis.extract.0"]["requirements"][0]["quote"] = "Выдуманный текст"
        self.responses["patch"] = {"operations": [{"op": "replace", "path": "/requirements/0/quote", "value": quote}]}
        llm = FakeLLM(self.responses)
        out = self.run_stage(llm)
        self.assertFalse(out.flags)
        self.assertTrue(g0(self.workspace, out.artifacts).passed)
        self.assertEqual(sum(call["task_class"] == "patch" for call in llm.calls), 1)

    def test_duplicate_requirements_are_merged_by_normalized_text(self) -> None:
        duplicate = copy.deepcopy(self.responses["analysis.extract.0"]["requirements"][0])
        duplicate["id"] = "REQ-99"
        self.responses["analysis.extract.0"]["requirements"].append(duplicate)
        self.run_stage(FakeLLM(self.responses))
        analysis = Analysis.model_validate_json((self.workspace / "spec/analysis.json").read_text())
        self.assertEqual(len(analysis.requirements), 9)

    def test_revision_is_bounded_and_unsolved_quote_is_flagged(self) -> None:
        self.responses["analysis.extract.0"]["requirements"][0]["quote"] = "Выдуманный текст"

        llm = FakeLLM(self.responses)
        out = self.run_stage(llm)
        self.assertTrue(out.flags)
        self.assertEqual(sum(call["task_class"] == "patch" for call in llm.calls), 2)
        self.assertFalse(g0(self.workspace, out.artifacts).passed)

    def test_common_gherkin_keyword_formatting_is_normalized(self) -> None:

            ac["gherkin"] = ac["gherkin"].replace("Дано ", "Дано: ").replace("Когда ", "Когда: ").replace("Тогда ", "Тогда: ")
        out = self.run_stage(FakeLLM(self.responses))
        self.assertFalse(out.flags)
        self.assertNotIn("Дано:", (self.workspace / "spec/analysis.json").read_text())

        self.assertTrue(out.flags)

        template = self.responses["analysis.lint.0"]["questions"][0]

        self.responses["analysis.lint.0"]["issues"].append({"id": "ISS-04", "req": "REQ-02", "characteristic": "полнота", "problem": "Девятое уточнение", "action": "Q-09"})
        self.run_stage(FakeLLM(self.responses))
        analysis = Analysis.model_validate_json((self.workspace / "spec/analysis.json").read_text())
        self.assertEqual(len(analysis.questions), 8)

    def test_repair_cannot_change_unrelated_fields(self) -> None:
        self.run_stage(FakeLLM(self.responses))
        path = self.workspace / "spec/analysis.json"
        before = path.read_bytes()
        llm = FakeLLM({"patch": {"operations": [{"op": "replace", "path": "/profile/platforms/0", "value": "mobile"}]}})
        inp = StageInput(run_id="test", stage="analysis", attempt=1, requirements=self.requirements, artifacts={"analysis": "spec/analysis.json"})

        self.assertTrue(result.flags)
        self.assertEqual(before, path.read_bytes())

    def test_window_splitting_and_delimiter_injection(self) -> None:
        llm = FakeLLM({"req_extract": {"requirements": [], "noise": []}}, context_window=8192)
        items = [{"id": f"S-{i:03d}", "text": "x" * 2000} for i in range(10)]

## Короткий отчёт

Сделано: Реализованы извлечение, дедупликация, каркас домена, lint, A0/A1/A2, синтез и две ревизии. A0/A1 продолжает контрольную точку без повторного извлечения; A2 переводит defaults в ASM. Неизвестные роли ограничены внутренней схемой; однозначные подписи нормализуются в объявленные ID. Бюджет возвращает частичный результат.

Как проверить: команды выше; результаты общих тестов — в TESTS.md, оценка настоящей модели — в EVAL.md.

Допущения и ограничения: Стабильные ID нельзя менять при ремонте. Патч вне находки отклоняется без изменения артефакта. Нормализация Gherkin исправляет только двоеточия после ключевых слов шагов. Общие модули платформы не создаются; до их появления применяются локальные дублёры.
